```haskell
-- 화면마다 dayjs().format('…') 이 흩어져 같은 날짜가 다르게 찍힌다: 허용 표기를 타입으로 닫는다
-- dayjs 는 포맷 문자열이 곧 키이자 값 — 이름을 붙인 매핑 객체는 같은 정보를 두 번 든다
formats :: [Format]                                   -- 단일 출처 (DATE_FORMATS as const)
formats = [ "YYYY-MM-DD", "YYYY-MM-DDTHH:mm:ss", "HH:mm", "YYYY년 M월", "YY년 M월 D일"
          , "YYYY.MM.DD", "YY.MM.DD", "YYYY.MM.DD(ddd)", "YYYY.MM.DD HH:mm" ]

type DateFormat = { f :: Format | f ∈ formats }       -- 배열에서 파생된 닫힌 union, 따로 손댈 곳 없음

formatDate :: Date -> DateFormat -> String            -- 유일한 진입점
formatDate d f = format (dayjs d) f

-- 새 표기는 여러 화면에서 통일이 필요할 때만 formats 에 넣는다. 일회성('M.D')은 raw 로
-- 이 기준이 없으면 배열이 잡동사니 enum 이 된다

-- 구멍: raw 우회는 타입이 못 막는다 (no-restricted-syntax 규칙 전까지 단일 출처는 관행)
-- 구멍: "ddd" 는 locale "ko" 없이 Mon·Tue 로 조용히 샌다 — 런타임 전제를 가진 유일한 항목
```
