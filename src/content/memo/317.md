```haskell
-- expect.extend: Response 의 본문이 Zod 스키마에 맞는가
toMatchSchema :: Response -> Schema -> IO MatcherResult
toMatchSchema res schema = do
  body <- json res
  r    <- safeParseAsync schema body
  pure MatcherResult { pass = success r, message = "" }
```
