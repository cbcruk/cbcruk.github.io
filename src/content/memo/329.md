```haskell
-- Apps Script 로 캘린더 예약: API 두 개
available :: [Busy] -> [Slot] -> [Slot]
available busy = filter (not . overlapsAny busy)    -- 근무 요일·시간 슬롯 − Calendar.Freebusy.query

book :: Slot -> IO Event
book slot = do
  busy <- freebusy slot                             -- 계산할 때와 예약할 때 두 번 조회 — 그 사이에 일정이 생길 수 있다
  unless (null busy) (throw "Timeslot not available")
  createEvent (getCalendarById cal) title slot { sendInvites = True }   -- 기본 false: 안 주면 초대 메일이 안 간다
```
