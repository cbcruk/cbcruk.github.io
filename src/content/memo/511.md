![inArray(authors.id, [1, 2, 3]) 은 WHERE id IN (1, 2, 3) 이 된다. 빈 배열을 넘기면 에러 없이 WHERE false 가 되어 빈 결과가 돌아오지만 쿼리 왕복은 한 번 생긴다. 그래서 배열이 비었으면 쿼리 없이 바로 [] 를 돌려준다.](/svg/drizzle_in_array.svg)

```haskell
-- Drizzle 에서 IN 절 — 아래 inArray 는 drizzle-orm 소스의 동작 그대로다
inArray :: Column a -> [a] -> SQL Bool
inArray _   [] = sql "false"                   -- 빈 배열은 에러가 아니라 false 조건
inArray col xs = col `in'` xs                  -- WHERE id IN (1, 2, 3)

authors ids = select authorsTable `where'` inArray (id authorsTable) ids

-- 빈 배열이면 쿼리 왕복을 아낀다
authors' [] = pure []
authors' ids = authors ids
```
