![숫자만 남긴 뒤 길이와 02 로 시작하는지로 그룹 자릿수를 고른다. 11자리는 3-4-4, 02 로 시작하는 10자리는 2-4-4, 그 밖의 10자리는 3-3-4, 02 로 시작하는 9자리는 2-3-4. 나머지는 Invalid. replace 는 매치된 부분만 바꾸고 나머지를 뒤에 남기므로 그룹 자릿수의 합이 길이와 같아야 한다.](/svg/phone_format_groups.svg)

```haskell
-- 전화번호: 숫자만 남기고 (길이, 02 접두) 로 그룹 자릿수를 고른다
groups :: String -> Maybe [Int]
groups ds = case (length ds, "02" `isPrefixOf` ds) of
  (11, _)     -> Just [3, 4, 4]
  (10, True)  -> Just [2, 4, 4]
  (10, False) -> Just [3, 3, 4]
  (9,  True)  -> Just [2, 3, 4]
  _           -> Nothing                         -- 'Invalid phone number'

format :: String -> Maybe String
format raw = intercalate "-" . splitPlaces' ds <$> groups ds
  where ds = filter isDigit raw
-- replace 는 매치된 곳만 바꾸고 나머지는 뒤에 남긴다 → sum groups == length ds 여야 한다
```
