```haskell
-- MDIR 스타일 인터페이스로 SQLite DB 를 탐색하는 도구 — 붙일 자리가 셋
data Frontend = Telescope        -- 순수 Neovim. :StoryBrowse · :StorySearch
              | VSCodeExtension  -- Activity Bar + Tree View + Quick Pick
              | BunExecutable    -- bun build --compile: DB + 웹서버 + 프론트엔드 한 파일 (~100MB)

browse :: Frontend -> SQLite -> UI
```
