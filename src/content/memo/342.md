![스프레드시트가 커서 보기 힘들 때 첫 시트의 모든 값을 CSV 로 만들어 구글 드라이브에 파일로 올린다. 셀에 쉼표·줄바꿈·큰따옴표가 있으면 join(",") 만으로는 열이 밀리므로, 셀마다 큰따옴표로 감싸고 안의 " 는 "" 로 겹친다. 행은 줄바꿈으로 잇고 DriveApp.createFile 로 만든 파일의 URL 을 로그로 찍는다.](/svg/sheet_to_csv_drive.svg)

```haskell
-- 큰 시트를 CSV 파일로 드라이브에 올려 확인한다 (Apps Script)
quote :: String -> String
quote v = "\"" ++ concatMap (\c -> if c == '"' then "\"\"" else [c]) v ++ "\""   -- RFC 4180: 감싸고, " 는 "" 로

csv :: [[String]] -> String
csv = intercalate "\n" . map (intercalate "," . map quote)   -- join(',') 만이면 쉼표·줄바꿈에서 열이 밀린다

saveSpreadsheetDataAsCSV = do
  rows <- getValues . getDataRange . head =<< getSheets activeSpreadsheet
  file <- createFile "spreadsheet_data.csv" (csv rows) MimeCSV
  log ("파일이 생성되었습니다: " ++ url file)
```
