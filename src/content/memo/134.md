```haskell
-- 네이티브 select 를 버리지 않고 셋만 손댄다 (동작은 네이티브가)
style :: Select -> Css
style s = inheritFont          -- font · letter-spacing · word-spacing: inherit
       <> customArrow          -- appearance: none + 직접 그린 화살표
       <> stateStyles s        -- :focus · :required · :invalid
```
