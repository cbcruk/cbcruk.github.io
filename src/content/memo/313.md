```haskell
-- 대화 상자는 무엇을 막느냐로 갈린다
data Scope  = System | Application | Document     -- 막는 범위: 전체 → 프로그램 → 부모 창(macOS 시트)
data Dialog = Modal Scope | Modeless              -- Modeless: 열어 둔 채 작업 (툴바)

blocks :: Dialog -> Window -> Bool
blocks (Modal System)      _ = True
blocks (Modal Application) w = sameApp w
blocks (Modal Document)    w = isParent w
blocks Modeless            _ = False

-- 모달 확인은 습관으로 눌린다 → 경고 대신 실행 취소
delete :: Item -> IO Undo     -- confirm :: Item -> IO Bool 이 아니라
```
