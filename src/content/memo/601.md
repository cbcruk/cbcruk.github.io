![OS 웹뷰가 주는 통로는 둘뿐이고 둘 다 문자열만 나른다. Go 는 JS 코드를 문자열로 넘겨 실행시키고(Eval), 페이지는 postMessage 로 문자열을 보낸다. 페이지에서 함수를 부르면 id 를 하나 만들어 Promise 를 걸어두고, id·함수 이름·인자를 JSON 문자열로 보낸다. Go 는 그 함수를 따로 실행한 뒤, 같은 id 와 결과를 넣은 onReply 호출을 Eval 로 돌려보낸다. 페이지는 id 로 기다리던 Promise 를 찾아 끝낸다.](/svg/webview_bridge.svg)

```haskell
-- 웹 페이지를 터미널 프로그램처럼 돌린다 — 브라우저를 싣지 않고 OS 웹뷰를 빌린다 (webframe)
webframe :: URL -> Process
webframe = hiddenWebview >>> fullSpeed >>> connect
  -- connect: console → stdout, 예외 → stderr, stdin → eval — 전부 아래 두 통로 위에 있다

-- OS 웹뷰가 주는 통로는 둘뿐이고, 둘 다 문자열만 나른다
eval :: String -> IO ()                   -- Go → JS: 코드를 실행시킨다 (돌려받는 값은 버린다)
post :: String -> IO ()                   -- JS → Go: 문자열을 보낸다

-- 같은 두 통로, OS 마다 이름만 다르다. Go 는 C 컴파일러 없이 라이브러리를 열어 부른다 (purego)
channels :: OS -> (Eval, Post)
channels MacOS   = ( "evaluateJavaScript:completionHandler:"          -- objc_msgSend
                   , "window.webkit.messageHandlers.__webview__.postMessage" )
channels Linux   = ( "webkit_web_view_evaluate_javascript"            -- dlopen 한 C 함수
                   , "window.webkit.messageHandlers.__webview__.postMessage" )
channels Windows = ( "ICoreWebView2::ExecuteScript"                   -- COM vtable 의 함수 포인터
                   , "window.chrome.webview.postMessage" )

-- 둘을 엮으면 함수 호출이 된다. 요청과 응답을 짝짓는 열쇠는 id 하나
call :: Name -> [Json] -> Promise Json     -- Bind "myFn" goFn 하면 window.myFn 이 이것이다
call name params = do
  id <- randomId                           -- _promises[id] 에 Promise 를 걸어둔다
  post (encode { id, method = name, params })
  awaitReply id                            -- onReply 가 id 로 찾아 resolve·reject 하고 지운다

-- Go 쪽: 받은 문자열을 풀어 함수를 찾고, 결과를 다시 JS 코드 문자열로 돌려보낸다
onMessage :: String -> IO ()
onMessage body = do
  { id, method, params } <- decode body
  forkIO $ do                              -- 고루틴: UI 스레드를 막지 않는다
    (status, result) <- (bindings ! method) params
    onMain $ eval ("window.__webview__.onReply(" <> id <> ", " <> status <> ", " <> result <> ")")
                                           -- 웹뷰는 UI 스레드에서만 만질 수 있다

-- webframe 의 hooks.js 는 console·예외를 이 call 로 Go 에 보낸다.
-- 고루틴마다 따로 돌아 순서가 섞이므로, 응답이 올 때까지 다음 묶음을 모아둔다 (한 번에 하나만)
```
