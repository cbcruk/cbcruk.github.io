![Miss: 브라우저 → 엣지 → 오리진(S3)까지 가서 새로 받는다. 엣지 캐시가 비어 있었을 뿐 최신이라는 뜻은 아니다. Hit: 브라우저 → 엣지에서 멈추고 오리진에는 가지 않은 채 캐시본을 돌려준다 — 구버전이 서빙되는 지점이다. RefreshHit: 브라우저 → 엣지 → 오리진에 재검증 → 304 → 엣지가 캐시본을 돌려준다 — max-age=0 의 정상 동작이고 HTML 의 목표 상태다. 엣지의 TTL 타이머는 POP·경로별로 독립이라 새 화면과 구 화면이 재현성 없이 섞인다.](/svg/cloudfront_xcache_paths.svg)

```haskell
-- SPA → Next.js SSG 로 옮기니 배포해도 구버전 HTML 이 나온다: 라우트마다 HTML 이 독립 TTL 로 엣지에 남는다
-- SPA 는 HTML 이 index.html 하나라 진입점만 무효화하면 됐다
-- Cache-Control 이 없으면 엣지는 Default TTL 24h, 브라우저는 휴리스틱 (Date − Last-Modified) × 10%
-- 그 동안 엣지는 오리진에 가지 않는다 — 로컬에서 재현 안 돼도 문제다

cacheControl :: Object -> CacheControl
cacheControl (Static _hash) = "public,max-age=31536000,immutable"  -- 이름에 해시: 내용이 바뀌면 이름이 바뀐다
cacheControl _              = "public,max-age=0,must-revalidate"   -- HTML·데이터: 매 요청 재검증, 304

-- 어느 층이 구버전을 들고 있나 — 안쪽부터 (오리진이 구버전이면 바깥은 볼 이유가 없다)
data XCache = Miss | Hit | RefreshHit

stale :: Layer -> Verdict
stale (Origin lastModified) | lastModified < deployedAt = NotDeployed
stale (Edge Hit Nothing _)                    = NoHeader      -- 전환 최초 1회 '/*' 무효화
stale (Edge Hit (Just "max-age=0") ageGrows)  = MinimumTtl    -- Cache Policy 가 오리진 헤더를 덮는다
stale (Edge RefreshHit _ _)                   = Fine          -- HTML 의 목표 상태
stale Browser                                 = Heuristic     -- 무효화가 못 닿는다, 만료를 기다린다
                                                                --   헤더 적용 이후 배포분부터는 재발하지 않는다

-- 배포는 에셋 → HTML → 무효화 순
deploy = sync static [] >> sync html ["--delete"] >> invalidate "/*"
-- HTML 이 먼저 가면 청크 없는 창이 생긴다. 이 순서면 HTML 단계가 실패해도 "새 에셋 + 구 HTML" 로 정상 동작한다
-- 구 해시는 남겨야 열린 탭의 lazy chunk 가 404 가 안 난다
-- 헤더 적용 후에는 HTML 이 max-age=0 이라 엣지가 매 요청 재검증한다 — 무효화는 즉시 반영을 위한 보험일 뿐이다
-- --acl public-read 는 둘 다에 (BucketOwnerEnforced 면 빼고 OAC)

-- 확인: curl -sI https://example.com/a | grep -iE 'cache-control|x-cache|age'
```
