```haskell
-- 한 켤레는 두 축으로 정해진다: 오늘 메뉴의 목적 × 시즌의 시기
shoe :: Phase -> Menu -> Shoe
shoe phase menu = pick (plate phase) (purpose menu)

-- ① 오늘 달릴 메뉴의 목적을 먼저 확인한다
data Purpose = Race | SpeedAndForm | PaceAndDistance | DailyJog

purpose :: Menu -> Purpose
purpose RaceDay      = Race
purpose Interval     = SpeedAndForm      -- 질주·완주 반복, 심폐 기능
purpose RacePractice = SpeedAndForm      -- 실전 페이스·집단주를 가정한 고강도
purpose Sprint       = SpeedAndForm      -- 전력에 가까운 단거리, 동작 만들기
purpose LongRun      = PaceAndDistance
purpose FastJog      = PaceAndDistance   -- 경쾌한 페이스로 거리를 쌓는다
purpose PaceRun      = PaceAndDistance   -- 목표 레이스 페이스를 일정하게
purpose SlowJog      = DailyJog          -- 포인트 훈련 다음 날 피로 빼기

-- 가장 자주 신는 한 켤레
needs :: Purpose -> [Trait]
needs DailyJog = [Cushion, Support]

-- ② 목표 레이스에서 거꾸로 세어 시기를 나눈다 (期分け)
data Phase = Base | Build | RaceSeason

plate :: Phase -> Plate
plate Base       = NonCarbon                   -- 봄·트랙: 다리 근력과 접지 감각
plate Build      = NonCarbon `towards` Carbon  -- 여름 합숙: 토대를 만들며 가을 로드에 맞춰 점점 익힌다
plate RaceSeason = Carbon                      -- 가을·겨울 로드: 연습에서도 실전용을 신는 일이 늘어난다

-- 기초기에 일부러 반발이 적은 신발을 고르는 이유: 신발에 기대지 않고 '자기 다리'를 단련한다.
-- 발가락으로 땅을 제대로 잡는 감각, 다치기 어려운 토대 (花田勝彦, 早稲田大学競走部 감독)
-- 원문의 수준별(アスリート 등) 모델 추천 표는 이미지라 옮기지 않았다
```
