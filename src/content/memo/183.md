```haskell
-- psd.js: PSD 를 트리로 읽는다 (Node · 브라우저)
data Document = Document { size :: Size, colorMode :: Mode, depth :: Int
                         , merged :: Image, comps :: [Comp], tree :: [Node] }

data Node = Group Props [Node]
          | Layer Props (Maybe Text) (Maybe VectorMask)

data Props = Props { name :: String, bounds :: Rect, visible :: Bool, opacity :: Double }
data Text  = Text  { content :: String, font :: String, fontSize :: Double, color :: Color }  -- psd-enginedata
```
