```haskell
-- 파일질라 서버로 FTP 열기 (Windows)
setup :: Windows ()
setup = install FileZillaServer
     >> addUser                 -- Edit → Users → Add
     >> allowFirewall           -- 제어판 → Windows Defender 방화벽 → 허용되는 앱 → 파일질라 서버 추가
```
