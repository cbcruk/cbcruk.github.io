```haskell
-- 응답 하나 = 셸 하나 + 기다리던 경계마다 조각 하나. 마지막 경계까지 응답은 열려 있다
stream :: Tree -> [Chunk]
stream t = shell t : [ segment b <> swap b | b <- readyOrder (pending t) ]   -- 준비된 순서, 문서 순서가 아니다
  -- 마지막 조각 뒤에야 응답이 닫히고 브라우저 파싱이 끝난다 → defer 스크립트는 가장 느린 경계를 기다린다

-- 셸에서 경계 자리
boundary :: Boundary -> Chunk
boundary b | ready b   = "<!--$-->"  <> html b                                        <> "<!--/$-->"
           | failed b  = "<!--$!-->" <> fallback b                                    <> "<!--/$-->"  -- 클라이언트가 다시 그린다
           | otherwise = "<!--$?--><template id=\"B:" <> n b <> "\"></template>" <> fallback b <> "<!--/$-->"
  -- ready 라도 뒤로 뺀다(셸엔 $? 자리, 내용은 segment+swap): 500B 넘는 경계가 이미 쓴 것과 합쳐 12,800B(progressiveChunkSize) 를 넘을 때
  --   · 스타일시트를 기다려야 할 때 · defer 일 때. 셸이 너무 커져 첫 화면이 늦어지지 않게

segment b = "<div hidden id=\"S:" <> n b <> "\">" <> html b <> "</div>"   -- 화면 밖에 먼저 붙는다
swap    b = "<script>$RC(\"B:" <> n b <> "\",\"S:" <> n b <> "\")</script>" -- $RC 정의는 첫 swap 앞에 한 번 인라인으로

-- $RC: 바로 바꾸지 않는다. 표시를 $~ 로 바꿔 대기열에 넣고 한꺼번에 바꾼다($RV)
reveal :: Ms -> Maybe Ms -> Delay
reveal _   Nothing               = nextFrame                 -- 아직 한 번도 안 바꿨다
reveal now (Just _) | 2000 < now && now < 2300 = 2300 - now  -- 이 구간이면 2.3초에 맞춘다
reveal now (Just last)           = last + 300 - now          -- 지난 바꾸기 300ms 뒤
  -- $RV: 표시 사이 fallback 노드를 지우고 S:n 의 자식을 옮겨 넣고 표시를 "$" 로. _reactRetry 가 있으면 다음 프레임에 그 경계를 하이드레이트
  -- B:n 이 없으면(그 사이 지워졌다) S:n 도 지운다

-- 언제 흘려보내나 (renderToPipeableStream)
send :: Reader -> Hook
send Person  = onShellReady   -- 셸이 되면 pipe. 경계는 오는 대로
send Crawler = onAllReady     -- 다 풀린 뒤 pipe. 작은 경계는 셸 안에 "<!--$-->" 로, 큰 경계는 여전히 뒤로 빠져 $RC 로 끼운다
  -- 셸조차 실패하면 onShellError
  -- pipe: 버퍼가 차면 멈췄다 'drain' 에 이어 쓴다. 'close'·'error' 면 렌더를 취소. pipe 는 한 번만
```
