![compose(square, double)(3) 은 오른쪽부터 double 을 먼저 해 6, 그다음 square 로 36 이다. pipe(double, square)(3) 은 왼쪽부터 같은 순서라 역시 36. curry 는 인자가 함수의 length 만큼 모일 때까지 함수를 돌려줘서 add(1)(2)(3) 이 6 이 된다. 인자를 다 받기 전의 함수가 부분 적용이고, 데이터를 마지막 인자로 두면(data last) 부분 적용한 함수를 그대로 조합에 끼울 수 있다 — 변수 없이 함수만 잇는 포인트 프리 스타일이 된다.](/svg/curry_pipe_compose.svg)

```haskell
-- 작은 함수를 이어 붙여 새 함수를 만든다
compose :: [a -> a] -> a -> a
compose = foldr (.) id                 -- 오른쪽부터: compose [f, g, h] x = f (g (h x))
pipe :: [a -> a] -> a -> a
pipe = foldl (flip (.)) id             -- 왼쪽부터

compose [square, double] 3 == 36
pipe    [double, square] 3 == 36

-- curry: 인자가 fn.length 만큼 모일 때까지 함수를 돌려준다 (Haskell 은 원래 이렇다)
add a b c = a + b + c
add 1 2 3 == ((add 1) 2) 3 == 6
double = multiply 2                    -- 부분 적용
halve  = flip divide 2                 -- data last: 데이터를 마지막 인자로 두면 조합에 그대로 끼운다
pipeline = square . halve              -- 포인트 프리: 변수 없이
```
