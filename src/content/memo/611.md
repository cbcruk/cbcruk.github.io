```haskell
-- 장면 하나 = console.log 한 번
frame :: Grid -> Maybe (Format, [Css])
frame g | changedCells g == 0 = Nothing                 -- log 도 clear 도 부르지 않는다
        | otherwise           = Just (assemble (map spans (rows g)))
  -- Grid: 칸마다 글자 · 글자색 · 배경색(0xRRGGBBAA) · 속성. React 리컨실러 → Yoga 레이아웃이 채운다
  -- spans: 행 해시가 바뀐 행만 다시 훑어 같은 스타일 칸을 묶는다
  -- assemble: 묶음마다 "%c" ++ 글자, css 인자 하나. \n 양쪽 스타일이 같으면 한 묶음으로 잇는다
  --           빈칸은 배경이 같은 이웃에 흡수, 묶이지 않은 \n 앞에선 %c "" 로 스타일을 끊는다

-- 장면 바꾸기: 지우지 않고 아래에 덧붙인다
emit :: Count -> Frame -> IO ()
emit n f | n == 0 || n >= 300 || (n >= 100 && idle 1000) = clear >> log (curtain <> f)
         | otherwise                                        =          log (curtain <> f)
  -- curtain: margin-top:100vh 짜리 폭 0 묶음. 앞 장면을 위로 밀어 짧은 장면 아래로 옛 꼬리가 안 보이게
  --          Firefox · Safari 는 인라인의 세로 margin 을 무시해 line-height:100vh
  -- 매 장면 clear 하면: DevTools 는 clear 는 바로, 새 메시지는 50ms throttle 로 그린다 → 빈 콘솔이 깜빡
  -- 간격은 66ms 이상(~15fps): 더 빠르면 throttle 이 다시 걸려 장면이 버려진다
  -- clear 직후엔 한 번 110ms 기다린다: "Console was cleared" 줄이 즉시 슬롯을 먹어 다음 장면이 50ms 타이머로 밀린다
  -- 행 해시는 같은데 다시 그려야 하면 끝에 ​ 를 번갈아 붙인다: 같은 메시지는 반복 배지로 접혀 옆으로 밀린다

-- 글자를 서식 문자열에 넣을 때 DevTools 가 망치는 것
escape :: Text -> Text
escape = replace "%" "%%"                    -- "100%" 가 %c 인자를 먹지 않게
       . insertWordJoiner ["://", "www.", "path:line"]   -- ⁠: 링크로 바뀌며 색을 덮어쓰지 않게
  -- css 는 개별 속성만(font: 같은 축약 X): DevTools 가 묶음의 개별 속성을 줄마다 복사한다
  -- 배경 묶음엔 padding-right:1px; margin-right:-1px: Chromium 이 묶음마다 픽셀을 맞춰 생기는 1px 틈을 덮는다
  -- 입력은 콘솔에 없어서 페이지의 숨긴 <textarea> 로 받는다
```
