`gulp.src` 파일 리스트 디버깅

```javascript
// gulpfile.mjs — gulp 5 · gulp-debug 5 · glob 14 (gulp-debug 5 는 ESM 전용)
import gulp from 'gulp'
import debug from 'gulp-debug'
import through2 from 'through2'
import gulpif from 'gulp-if'
import { globSync } from 'glob'

// 1. gulp-debug (가장 간단)
gulp.src('src/**/*.js').pipe(debug({ title: 'Files:' }))

// 2. through2 (상세 정보)
gulp.src('src/**/*.js').pipe(
  through2.obj((file, enc, cb) => {
    console.log(file.relative)
    cb(null, file)
  })
)

// 3. glob 패턴 사전 확인
console.log(globSync('src/**/*.js'))

// 조건부 디버깅: `NODE_ENV=debug gulp build`
gulp.src('src/**/*.js').pipe(gulpif(process.env.NODE_ENV === 'debug', debug()))
```
