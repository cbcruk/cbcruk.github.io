![Next.js 의 prefetchViaDom 은 같은 href 의 prefetch·preload link 나 script 가 이미 있으면 건너뛰고, 없으면 link rel=prefetch 를 만들어 head 에 붙인다. 속성 순서가 의도적이라 href 를 맨 마지막에 넣는다 — href 를 넣는 순간 요청이 나가므로 그 전에 as·crossOrigin·onload·onerror 가 준비돼 있어야 한다. mini-css-extract-plugin 은 CSS chunk 의 link 가 실패하면 ChunkLoadError 를 만들고 그 link 태그를 DOM 에서 지운 뒤 reject 한다.](/svg/next_prefetch_css_chunk.svg)

```haskell
-- Next.js prefetchViaDom: 이미 있으면 건너뛰고, 없으면 link 를 붙인다
prefetch :: Href -> IO ()
prefetch href = do
  exists <- querySelector ("link[rel=prefetch|preload][href^=" ++ href ++ "], script[src^=…]")
  unless exists $ do
    l <- createElement "link"
    set l [As, Rel "prefetch", CrossOrigin, OnLoad resolve, OnError (reject "Failed to prefetch")]
    set l [Href href]                   -- 맨 마지막: href 를 넣는 순간 요청이 나간다
    append head l

-- mini-css-extract-plugin: CSS chunk 실패
onCssError :: Event -> IO ()
onCssError e = remove (linkTag e) >> reject (ChunkLoadError "CSS_CHUNK_LOAD_FAILED" (href e))
```
