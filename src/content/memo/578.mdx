```haskell
-- useMemo 로 감쌌는데 렌더가 안 빨라진다
-- 재사용 조건: deps 가 같은 물건(Object.is)인가. 안을 들여다보지 않는다
reused :: Deps -> Deps -> Bool
reused prev next = and (zipWith objectIs prev next)

-- 렌더 중에 만든 값은 내용이 같아도 매번 다른 물건
reused [map transform data] [map transform data] == False

-- 고침: 넘기는 쪽에서 참조를 붙잡는다
items = useMemo (\() -> map transform data) [data]

-- 단계마다 추론이 아니라 관찰로 넘어간다
-- 1 느린가      Profiler 렌더 ms — 병목이 아니면 멈춘다, 5 의 기준선
-- 2 재사용되나  console.count "calc" 가 렌더 따라 느는가 (dev 는 렌더당 2 가 정상)
--               코드를 못 고치면 CDP Profiler.takePreciseCoverage callCount (부를 때마다 0 으로)
-- 3 왜          useRef 로 objectIs prev items 를 찍는다 · "Record why each component rendered"
-- 4 누가        같은 비교를 부모로 거슬러 올린다
-- 5 고쳐졌나    횟수와 ms 가 둘 다 줄었나. 횟수만 줄면 병목이 다른 곳 → 1
```
