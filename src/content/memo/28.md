```haskell
-- 캘린더 라이브러리: 상황으로 고른다 (위에서부터 매칭)
choose :: Need -> Library
choose FullyCustomUI = ReactAria          -- Adobe headless: 로직·접근성만, 일정 표시는 직접
choose QuickBasics   = ReactBigCalendar   -- 구글 캘린더 스타일, moment/date-fns/dayjs 선택 · 타입 아쉬움
choose ModernLight   = ScheduleX          -- React/Vue/Angular, 드래그·리사이즈 · 레퍼런스 적음
choose Enterprise    = FullCalendar       -- 플러그인, 타임라인·리소스 뷰 · 일부 유료, 번들 큼

-- 날짜 선택만: react-day-picker · dayzed · react-datepicker
-- headless: use-lilius · h6s — 찾기 어렵다, 갖다 써 보면 안다
```
