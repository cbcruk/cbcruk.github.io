```haskell
-- SPA 는 새로고침이 없어 서버처럼 오래 산다: 한 브라우저 컨텍스트에서 같은 플로우를 수백 번
type Flow = Page -> Page      -- 시작 화면 = 끝 화면 (drawer 열고 닫기)

-- 같은 자리로 돌아왔으면 수도 같아야 한다. 늘었으면 누수다
leaks :: Flow -> Page -> Bool
leaks flow page = count after > count base
  where
    base  = iterate flow page !! 5                    -- 워밍업: 첫 루프에 코드·데이터를 받아 heap 이 뛴다
    after = iterate (tick 18 . flow) base !! 195      -- 루프마다 가짜 시계 18초

count :: Page -> (Listeners, Nodes)   -- GC 두 번 뒤 CDP Performance.getMetrics
-- heap 은 판정에서 뺀다: 앱과 무관하게 실행마다 흔들린다. 수는 첫 루프부터 같은 만큼 오른다
-- 클릭은 Playwright, 측정은 CDP — CDP 세션은 Chromium 에서만 열린다
```
