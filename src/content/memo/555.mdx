![링크 클릭·폼 제출·프로그램 호출·history.pushState()·뒤로·앞으로까지 떠나는 모든 경로가 navigate 이벤트 하나로 모인다. 블로커가 켜져 있을 때 navigationType 이 push·replace·reload 면 preventDefault() 로 아예 안 움직이게 막는다(사전 차단) — 이때 destination.index 는 -1 이다. traverse(뒤로·앞으로)는 사용자를 사이트에 가둘 수 없도록 preventDefault() 로 취소할 수 없어 이미 움직인 뒤다. 그래서 history.go(현재 index − destination.index) 로 되돌린다(사후 롤백) — destination.index 는 traverse 일 때만 목적지 커서 값을 준다. API 자체가 같은 선을 긋고 있어 블로커 하나로 통일할 수 없고 라우터 안에 두 갈래가 남는다.](/svg/nav_two_paths.svg)

```haskell
-- Navigation API 라우터에 이탈 방지("정말 나가시겠습니까")를 붙인다
-- 떠나는 경로는 전부 navigate 이벤트 하나로 모이고, 갈래는 navigationType 이 가른다
data NavType = Push | Replace | Reload | Traverse

block :: NavType -> Index -> Index -> Action
block Traverse cur dest = Go (cur - dest)   -- 사후 롤백: 뒤로·앞으로는 이미 움직였다
block _        _   _    = PreventDefault    -- 사전 차단: 아예 안 움직인다 (dest = -1)
-- traverse 는 preventDefault 로 못 막는다 — 사용자를 사이트에 가두지 못하게
-- destination.index 는 traverse 일 때만 값이 있고 나머지는 -1

-- history = 리스트 + 커서. go n 은 리스트를 안 건드리고 커서만 옮긴다
go :: Int -> History -> History
go n h | 0 <= i && i < length (entries h) = h { cursor = i }
       | otherwise                        = h                  -- 범위 밖이면 아무 일도 없다
  where i = cursor h + n

go 1 (back h) == h   -- 커서 C → B → C, 겉보기엔 아무 일도 없었다
-- 안 일어난 이동을 되돌리면 진짜로 움직인다 (이 패턴에선 이동이 없으면 navigate 가 안 뜬다)
-- 브라우저가 사용자를 위해 막아둔 걸 되돌리는 우회라 정말 붙잡아야 할 때만 쓴다

-- 확인: 블로커를 켜고 다섯 이동을 다 해본다
--   push·replace·reload → 안 움직인다 · 뒤로·앞으로 → 움직였다 제자리로
```
