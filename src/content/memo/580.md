```haskell
-- 운영시간 API 가 통째로 500, 프론트는 그걸 "선택 가능한 날짜 없음" 달력으로 그린다
-- 원인: to = 24:00:00 한 행 — DateTimeException: HourOfDay (0 - 23): 24
type Domain = (Seconds, Seconds)

mysqlTime, localTime :: Domain
mysqlTime = (-838:59:59, 838:59:59)       -- duration 타입이라 24:00:00 은 합법
localTime = (00:00:00, 23:59:59)          -- datetime.time 도 같다. Duration·string 은 넓다

-- 넓은 쓰기 / 좁은 읽기
timeBomb :: Domain -> Domain -> Bool
timeBomb write read = not (write `within` read)

-- 정규화 후보는 슬롯 계산에 넣어 본다
normalize :: Candidate -> Outcome
normalize (To "23:59:59") = LoseLastSlot  -- to 가 exclusive 면 23:30 슬롯이 사라진다
normalize (To "00:00:00") = ZeroMinutes   -- from > to 로 뒤집혀 예외 없이 영업 0분 — 지금보다 나쁘다
normalize PromoteDuration = NoLoss        -- 하위 계산을 전부 고친다

-- 순서: 감사(TIME_TO_SEC ≥ 86400 · < 0) → 정규화 결정 → 핫픽스
--       → 읽기 방어(TypeHandler: ≥ 1일이면 23:59:59) → 쓰기 제약(CHECK, 8.0.16+)
-- 읽기가 먼저: 쓰기부터 막으면 이미 든 행 때문에 여전히 500
-- 프론트는 별개: 실패(다시 시도) ≠ 빈 배열(휴무)

-- 같은 폭탄: 스키마끼리가 아니라 DB 정의역 위에 겹쳐 본다
timeBomb decimal20 jsNumber   -- DECIMAL(20,0)·BIGINT → 2^53
timeBomb zeroDate  localDate  -- DATE '0000-00-00'
```
