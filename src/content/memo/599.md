![정상이면 OPTIONS 가 LB 를 지나 앱에 닿고 앱이 204 와 access-control-allow-origin 을 돌려준다. 장애면 등록된 타깃이 0 이라 LB 가 CORS 헤더 없는 503 을 직접 만들어 돌려주고, 브라우저는 그걸 CORS error 로 표시한다.](/svg/cors_preflight_lb_503.svg)

```haskell
data Server = ALB | App
data Origin = Star | Origin String deriving Eq

data Preflight = Preflight
  { status       :: Int
  , server       :: Server
  , acao         :: Maybe Origin
  , allowHeaders :: [Header]
  }

data Cause = Infra Infra
           | Route                -- path 오타, ingress rule 불일치
           | Cors Cors
           | NotCors              -- CORS 아님. 본 요청의 본문·인증·에러를 본다
data Infra = NoTargets            -- 서비스 다운·배포 실패·CrashLoop
           | BadTarget            -- 크래시·OOM·포트 불일치
           | IdleTimeout          -- 느린 쿼리·외부 연동 지연
data Cors  = Unset                -- CORS 미설정
           | MissingAllowHeaders  -- access-control-allow-headers 에 요청 헤더가 빠짐
           | StarWithCredentials  -- credentials 와 * 는 병용 불가 — 오리진을 명시한다

-- 브라우저가 보는 것: ACAO 가 없으면 원인을 지운다
browser :: Preflight -> Either CORSError Preflight
browser p = maybe (Left CORSError) (const (Right p)) (acao p)

-- DevTools 에서 하는 것: 지워지기 전의 값을 본다
-- preflight 의 status 와 server 헤더 둘로 가른다 (server: awselb/2.0 이면 LB 가 직접 만든 응답)
diagnose :: Request -> Preflight -> Cause
diagnose _ (Preflight 503 ALB _ _)       = Infra NoTargets
diagnose _ (Preflight 502 ALB _ _)       = Infra BadTarget
diagnose _ (Preflight 504 ALB _ _)       = Infra IdleTimeout
diagnose _ (Preflight 404 App Nothing _) = Route
diagnose _ (Preflight _   App Nothing _) = Cors Unset
diagnose r (Preflight _ App (Just Star) _)
  | credentials r                        = Cors StarWithCredentials
diagnose r (Preflight _ App (Just _) hs)
  | not (headers r `isSubsetOf` hs)      = Cors MissingAllowHeaders
diagnose _ _                             = NotCors

-- LB 가 직접 만든 응답이다 — 앱 응답이 아니니 CORS 헤더가 붙을 수 없다 (acao = Nothing)
-- 타깃이 0 일 때만 503. 전부 unhealthy 면 그래도 보낸다
alb :: [Target] -> Request -> Response
alb [] _ = Response 503 ALB Nothing []
alb ts r = forward (if all unhealthy ts then ts else filter healthy ts) r
```
