```haskell
-- 원칙: 명확하고 구체적으로 · 생각할 시간을 준다
-- 전술: 하나씩 덧씌운다
prompt :: Task -> Prompt
prompt = fewShot examples          -- 예시로 스타일 고정
       . checkConditions           -- 조건이 충족됐는지 모델에게 먼저 확인
       . askStructured Json        -- JSON · HTML 같은 구조화된 출력
       . delimit "\"\"\""          -- 입력 경계: ``` · """ · <tag></tag>
```
