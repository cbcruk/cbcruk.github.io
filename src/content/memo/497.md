![styled-components 의 & 는 컴포넌트에 붙은 생성 클래스(예: .sc-a1b2)라 &-header 는 .sc-a1b2-header 가 되고, HTML 의 className="header" 와는 절대 맞지 않는다. 대신 그냥 .header 자식 선택자를 쓰거나, Header 를 styled 컴포넌트로 따로 만들어 Container 안에서 ${Header} 로 고른다 — styled() 로 만든 컴포넌트는 고정 클래스가 있어 선택자에 끼울 수 있다. 추천은 뒤쪽이다.](/svg/styled_components_child_selector.svg)

```haskell
-- styled-components 에서 BEM 식 &-header 를 HTML 에서 참조할 수 있나? 없다
amp :: Component -> Class
amp c = generatedClass c                    -- & = 해시된 생성 클래스
amp c ++ "-header" /= "header"              -- &-header 는 .sc-a1b2-header — className="header" 와 안 맞는다

-- 대신
plain   = ".header { color: red }"          -- 1 일반 클래스 선택자 + className="header"
byStyle = selector header ++ " { margin-bottom: 20px }"   -- 2 styled 컴포넌트로 (추천)

selector :: Styled -> Selector
selector (Styled c) = stableClass c         -- styled() 로 만든 컴포넌트는 고정 클래스가 있다 → ${Header}
```
