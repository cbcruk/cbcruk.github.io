```haskell
-- 원작 .exe 에서 "그 기능이 어디 있나"를 찾는다. 손에 쥔 실마리에 따라 길을 고른다
locate :: Clue -> Addr
locate (Text s)        = refs (le32 (find (cp949 s) dotData)) dotText
  -- 한국어 문구는 EXE 에 CP949 그대로 있다. 그 VA 를 .text 에서 4바이트 리터럴로 찾는다
  -- "값을 깎는다" @0x532890 → 참조 한 곳 0x481211 → 흥정 함수 0x4811E0
locate (Known v next)  = diff (snapshot (== v)) (after next (== v + 1))
  -- 날짜 19일인 바이트를 전부 잡고 → 여관에서 하루 → 20 이 된 것만, 근처에 년·월이 있는지 대조
  -- → 년·월·일 = 0x5A4D20/24/28 (i32 셋)
locate (Toggle a b)    = changed a b ∩ changed b a
  -- 값을 모르는 상태(지금 어느 화면인가): 메모리 전체 스냅샷 → 화면 A→B 에서 바뀐 것 ∩ B→A 에서 바뀐 것
locate (Array lo hi n) = longestRun (\x -> lo <= x && x <= hi) n
  -- 한 값은 노이즈지만, 시세 90~105 가 같은 간격으로 도시 수(226)만큼 늘어선 곳은 거의 하나다
  -- → 도시 구조체 0x5863B4, stride 92
locate (Plugin bin)    = immAfter (cmp eax version) bin
  -- 소스 없는 남의 .plugin: 버전을 4바이트로 묶어 CMP EAX,imm32 한 번에 비교한다
  -- 그 분기 안 MOV [x],imm32 · MOV reg,[addr] 에 게임 주소가 그대로 박혀 있다 (일본판 기준)

-- 데이터 주소 → 그걸 쓰는 코드 → 함수
toFunc :: Addr -> Func
toFunc a = enclosing (writers a ++ refs (le32 a) dotText)
  -- 런타임: Cheat Engine "find what writes/accesses" — 커널 디버거라 게임이 안 죽는다
  -- 정적: .text 에서 4바이트 검색. 예: 지금 도시 0x5B6154 를 쓰는 세 곳 앞이
  --       전부 call 0x49E3E0 ; cmp eax,2 → "들어가겠습니까?" 대화상자
  -- 상수도 앵커가 된다: 숙박 +30일 → 0x1E 를 쓰는 명령 중 날짜·소지금 주소 근처

-- 함수 하나 = 노트 하나 (note/TradeBargain-0x4811E0.md)
note :: Func -> Note
note f = Note { target    = "한국어판 CDS_95.EXE 1.2.0.0 · MD5 · ImageBase · 섹션"
              , howFound  = "찾은 길"
              , signature = "__thiscall int(Ctx*, int* price, int* outMode), ret 8"
              , tables    = "구조체 오프셋 · 반환값 매핑"
              , mistakes  = "처음엔 0x4A17F0 을 Player() 로 읽었다. 틀렸다" }
  -- 옮길 때 주석에 근거 주소(0x004xxxxx)를 달고 문구는 원본 그대로. 원본의 흠(안 걸리는 조건)도 그대로 두고 적는다
  -- 여러 함수를 훑어야 하는 화면 한 벌은 서브에이전트에 맡긴다: 정확한 주소·식·문구·표 값·부르는 곳

-- 밟은 것
--   문구가 서식으로 쪼개져 있으면 못 찾는다: "항구로" ✕ ← "[%s]의 %s%s 들어가겠습니까?" → 아는 주소에서 거꾸로
--   일본판(1.4.0.0)과 한국어판(1.2.0.0)은 주소가 전혀 다르다 → 노트마다 기준 EXE 를 맨 위에
--   세이브는 u16 년 + u8 월·일, 메모리는 i32 셋 → 세이브 오프셋에서 유도한 주소 가설은 전부 틀렸다
--   직접 짠 유저모드 HW 브레이크포인트(DR0/DR7)는 WOW64 게임을 죽였다 → 폐기하고 CE 를 Lua 로 자동화
--   대사가 하나 같아도 말투 셋·부관 유무 둘로 여러 벌인 때가 많다 → 문자열 앞뒤를 훑어 짝을 찾는다
```
