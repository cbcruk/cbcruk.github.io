type Props = {
  popover: 'auto' | 'hint' | 'manual'
  popovertarget: string

}

  hasBackdrop: boolean // CSS ::backdrop
  isPopoverOpen: boolean // CSS :popover-open
}

type Methods = {
  hidePopover: () => void
  showPopover: () => void
  togglePopover: () => void
}

  beforetoggle: () => void
  toggle: () => void
}
