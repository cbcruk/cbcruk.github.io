```haskell
-- 함수 호출에서 T 가 엉뚱하게 정해졌을 때: 추론은 딱 2단계, 타입 체크는 그 뒤 별개
infer :: [(Source, Target)] -> Map TVar Type
infer = fmap resolve . collect

-- ① 수집: source 와 target 을 나란히 walk 하다 bare T 를 만나면 그 자리 source 를 후보로
collect :: [(Source, Target)] -> Map TVar Candidates
data Candidates = Candidates { co :: [Type], contra :: [Type] }   -- 함수 매개변수 자리만 반변
--   후보가 안 되는 것: extends 바운드 · 기본값 · 조건부의 조건 · 상위 타입
--   조건부 target 은 조건을 평가하지 않고 두 가지에 다 맞춘다 → 체크는 다른 가지로 가 TS2345
--   union source 는 가지마다 따로 walk 하고, 후보는 다시 union 되지 않는다
--   A & B 에 42: 모호해서 기록하지 않는다 → unknown. A & X 는 차집합이 아니라 포획 최대화

-- ② 해소: 리스트마다 하나로 접는다. 결과는 리스트에 없는 타입일 수 없다
resolve :: Candidates -> Type
resolve (Candidates [] []) = declaredDefault `orElse` Unknown     -- 호출되는 f 의 기본값만 본다
resolve (Candidates cs ks)
  | fold cs <: fold ks = fold cs     -- 더 정밀한 공변이 이긴다
  | otherwise          = fold ks     -- 반변이 바운드를 못 지키면 공변으로 폴백

fold :: [Type] -> Type
fold ts
  | sameBaseLiterals ts      = union ts   -- f('a', 'b') = "a" | "b"
  | Just t <- strictSuper ts = t          -- f(dog, animal) = Animal
  | otherwise                = head ts    -- f(dog, cat) = Dog — Animal 도 Dog | Cat 도 아니다

-- 우회: 바운드에만 있는 변수는 후보가 없다 (V in T extends Map<K, V> → unknown)
--   extractor GetV<T> (union 인자 OK, 반환 캐스트 필요) | 매개변수로 드러내기 T & Map<K, V> (캐스트 없음, union 에서 깨짐)
```
