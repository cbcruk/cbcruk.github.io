![태스크는 콜백으로 결과를 돌려주는 함수다. 병렬 러너는 모두 한꺼번에 시작하고 결과가 올 때마다 쌓아 개수가 태스크 수와 같아지면 끝 콜백을 부른다 — 결과는 끝난 순서다. 직렬 러너는 하나를 시작하고 그 콜백 안에서 다음을 시작해, 인덱스가 끝에 닿으면 끝 콜백을 부른다 — 결과는 넣은 순서다.](/svg/task_runner_parallel_serial.svg)

```haskell
-- 태스크 = 결과를 콜백으로 돌려주는 함수
-- 두 실행기는 같은 부모(TaskRunner)에서 tasks·results 목록과 addTask 를 물려받고, run 만 다르다
type Task = (String -> IO ()) -> IO ()

-- 병렬: 한꺼번에 시작, 결과가 태스크 수만큼 모이면 끝 — 결과는 끝난 순서
parallel :: [Task] -> ([String] -> IO ()) -> IO ()
parallel ts done = do
  rs <- newRef []
  forM_ ts $ \t -> t $ \r -> do
    modify rs (++ [r])
    n <- length <$> read rs
    when (n == length ts) (done =<< read rs)

-- 직렬: 다음 태스크를 앞 태스크의 콜백에서 시작 — 결과는 넣은 순서
serial :: [Task] -> ([String] -> IO ()) -> IO ()
serial ts done = foldr (\t k rs -> t (\r -> k (rs ++ [r]))) done ts []
```
